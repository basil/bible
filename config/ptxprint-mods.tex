% An acrostic name of Lamentations opens its verse, so the verse milestone
% stands in the name's heading, where the source puts it; print the number
% beside the first line of the poem instead, as a hanging verse number is
% set. The ordinary rendering is kept, with its font and spacing: a milestone
% met in a qa paragraph is boxed, and the next q1 line sets the box: the
% line stage cuts a name from the opening of its q1 line, and the view
% indents only a line that follows a line of poetry, so a q1 line always
% follows. A name before the box is set, or the end of the book, is an
% error. A name that opens a chapter leaves the chapter number to drop into
% that q1 line, since a heading takes none, and its verse milestone came
% before the number, where PTXprint cannot omit or unhang verse one: do so
% here, as it does after a dropped number, rather than hang the number
% against the chapter figure.
\newbox\BibleAcrosticVerse
\begingroup
\catcode`\@=11
\gdef\BibleAcrosticMarker{qa}
\global\let\BiblePrintVerse\defaultprintverse
\gdef\defaultprintverse{%
  \ifx\m@rker\BibleAcrosticMarker
    \ifvoid\BibleAcrosticVerse\else
      \errmessage{Unplaced acrostic verse number}%
    \fi
    \global\setbox\BibleAcrosticVerse=\hbox{\BiblePrintVerse}%
  \else
    \BiblePrintVerse
  \fi
}
\addtohook{start}{q1}{%
  \ifvoid\BibleAcrosticVerse\else
    \ifdr@ppednumber
      \ifOmitVerseNumberOne
        \global\setbox\BibleAcrosticVerse=\box\voidb@x
      \else
        \box\BibleAcrosticVerse
      \fi
    \else
      % Match PTXprint's native hanging printer: right-align the captured
      % label and its prose spacing before the text without moving the text.
      \llap{\box\BibleAcrosticVerse}%
    \fi
    % The line no longer opens at its hanging position: set a verse that
    % opens it inline.
    \hangingversefalse
  \fi
}
\endgroup
\setbookhook{end}{LAM}{%
  \ifvoid\BibleAcrosticVerse\else
    \errmessage{Unplaced acrostic verse number at end of Lamentations}%
  \fi
}

% Put cross references and footnotes beside their source verses in the wide
% inner margin, using PTXprint's native side-aligned notes rather than a
% second scripture column. Its repeated runs adjust crowded note positions;
% the source reference and caller still identify their verse. This native
% marginal-note mode does not move overflow to the bottom of the page.
% Divide the 42 mm inner margin into a 6 mm gap,
% 24 mm of notes, and 12 mm clear at the binding. PTXprint's
% marginnoteswidth includes the gap, so the allocation is 30 mm, not 24 mm.
\newif\ifBibleMarginNote
\newbox\BibleHeadingNoteBox
\let\BiblePrimitiveVadjust\vadjust
\begingroup
\catcode`\@=11
\marginaln@te{x}
\marginaln@te{f}
\global\marginnotesgap=6mm
\global\marginnoteswidth=\dimexpr 24mm+\marginnotesgap\relax
\gdef\MarginNoteSide{inner}
\gdef\BibleCrossReference{x}
\gdef\BibleStartNote#1{%
  \BibleMarginNotetrue
  % A note with a caller is set below the text, not in the margin. Its
  % caller takes its actual width, followed by a small gap. Set this here:
  % PTXprint's size setup resets the minimum caller width.
  \ifx\them@rk\empty\else
    \BibleMarginNotefalse
    \NoteCallerWidth=0pt
    \NoteCallerSpace=1pt
  \fi
  \ifx\ch@pter\empty\BibleMarginNotefalse\fi
  \ifx\ch@pter\z@ro\BibleMarginNotefalse\fi
  % Peripheral scripture references keep their callers to identify the
  % quoted words, but use the same margin as scripture's references.
  \ifinperiph
    \def\BibleNoteKind{#1}%
    \ifx\BibleNoteKind\BibleCrossReference
      \BibleMarginNotetrue
    \fi
  \fi
  \ifBibleMarginNote
    % A note in a heading (a Psalm's title) is set inside the heading box,
    % which forbids hyphenation and minimizes lines; the narrow column needs
    % both as elsewhere.
    \ifhe@dings\hyphenpenalty=50 \exhyphenpenalty=50 \linepenalty=10 \fi
    \n@test@rt{#1}%
  \else
    % The front matter's ordinary footnotes go to the foot, in
    % separate paragraphs: PTXprint's separate notes reserve space and
    % can split long notes. The stylesheet's Left justification is for the
    % margin column; at the foot a note is justified like the text. This
    % local assignment is read by PTXprint's side-skip setup inside its note
    % box, a group nested in this one, and is undone when the note ends.
    \expandafter\let\csname #1:justification\endcsname\relax
    \expandafter\insert\csname note-#1\endcsname
  \fi
}
\gdef\BibleEndNote#1{%
  \ifBibleMarginNote
    % A note in a heading (a Psalm's title) ends inside the box PTXprint
    % keeps for the heading's notes, where \vadjust is not allowed. Hold what
    % the note would adjust, and attach it to the heading's line once that
    % box closes and the heading's paragraph resumes, as a verse's note is
    % attached to its line. PTXprint closes the note's own box before it
    % places the note, so the swap must outlive that group.
    \ifhe@dings\global\let\vadjust\BibleHeadingAdjust\fi
    \n@te@nd{#1}%
    \ifhe@dings
      \global\let\vadjust\BiblePrimitiveVadjust
      \aftergroup\BibleHeadingAttach
    \fi
  \fi
}
\gdef\BibleHeadingAdjust#1{%
  \global\setbox\BibleHeadingNoteBox=\vbox{\unvbox\BibleHeadingNoteBox #1}%
}
\gdef\BibleHeadingAttach{%
  \ifvoid\BibleHeadingNoteBox\else
    \ifhmode\BiblePrimitiveVadjust{\unvbox\BibleHeadingNoteBox}%
    \else\unvbox\BibleHeadingNoteBox\fi
  \fi
}
\expandafter\gdef\csname note-no-insert-f\endcsname{\BibleStartNote{f}}
\expandafter\gdef\csname note-no-insert-x\endcsname{\BibleStartNote{x}}
\expandafter\gdef\csname endnote-no-insert-f\endcsname{\BibleEndNote{f}}
\expandafter\gdef\csname endnote-no-insert-x\endcsname{\BibleEndNote{x}}
\endgroup

% A heading set in its own size keeps its own leading. PTXprint (3.1.3,
% paratext2.tex, \@s@tbaseline) computes a paragraph's baseline from the
% font size it finds through the style stack and caches the result under
% \f@ntstyle, the style it was asked about. A table row (ptx-tables.tex,
% \t@blerow) asks about the previous paragraph's marker through the row's
% stack, so a table that follows a title caches the body's leading as the
% title's, and every mt1 after it is set
% on body leading: its wrapped lines touch, and a large line after a small
% one has no room. Cache a baseline only when it was looked up without a
% stack: PTXprint passes the same style twice for that. Every paragraph
% passes \m@rker and \styst@k, so a paragraph's baseline is worked out
% each time from its font size.
\begingroup
\catcode`\@=11
\gdef\@s@tbaseline#1#2{%
  \getp@ram{baseline}{#1}{#2}\ifx\p@ram\relax
    \getp@ram{fontsize}{#1}{#2}\ifx\p@ram\relax\else
      \def\s@urce{calcn:\f@ntstyle}%
      \dimen0=\p@ram\le@dingunit
      \multiply\dimen0 by \LineSpaceBase \divide\dimen0 by 12
      \def\BibleBaselineStyle{#1}\def\BibleBaselineStack{#2}%
      \ifx\BibleBaselineStyle\BibleBaselineStack
        \setp@ram{baseline}{\f@ntstyle}{\the\dimen0}%
        \should@xist{}{\f@ntstyle}%
      \fi
      \baselineskip=\dimen0\advance \baselineskip by \ColAdjskip
    \fi
  \else
    \def\s@urce{store:#1,#2}%
    \baselineskip=\p@ram\fi
  \trace{j}{set baselineskip=\the\baselineskip (\s@urce)}}
\endgroup
