% PTXprint's wh hook already sets RTL. Make the intended quotation direction explicit.
\sethook{start}{wh}{\beginR}
\sethook{end}{wh}{\endR}

% Follow the Church of Greece's Daniel sequence internally (Susanna 0,
% Daniel 1-12, Bel and the Dragon 13), but use the named sections in running
% headers instead of exposing the suppressed numeric IDs.
\let\biblemarkverseNml\m@rkverseNml
\def\bibleDAG{DAG}
\def\m@rkverseNml{%
  \edef\biblebookid{\id@@@}%
  \ifx\biblebookid\bibleDAG
    \ifnum\ch@pter=0
      \mark{Susanna::\print@bleverse}%
      \trace{H}{mark: Susanna::\print@bleverse}%
    \else\ifnum\ch@pter=13
      \mark{Bel and the Dragon::\print@bleverse}%
      \trace{H}{mark: Bel and the Dragon::\print@bleverse}%
    \else
      \biblemarkverseNml
    \fi\fi
  \else
    \biblemarkverseNml
  \fi}
